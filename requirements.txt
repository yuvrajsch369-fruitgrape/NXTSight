# NXTSight — on-device pipeline dependencies
# Target runtime: Snapdragon X Elite NPU (Hexagon) via Qualcomm AI Hub / ONNX Runtime QNN EP

# Qualcomm AI Hub: compile, profile, and validate models for on-device NPU execution
qai-hub
qai-hub-models

# On-device inference runtime. onnxruntime-qnn ships ONNX Runtime with the
# QNN execution provider, which is how a compiled model actually reaches the
# Snapdragon Hexagon NPU — but it only publishes Windows/Linux-ARM wheels.
# Everywhere else (this dev Mac included) we install plain onnxruntime, and
# src/pipeline/runtime.py auto-detects which one is present at startup and
# falls back to the CPU provider when QNN isn't available.
onnx<1.20  # newer onnx defaults to an IR version this onnxruntime can't load
onnxruntime; sys_platform != "win32"
onnxruntime-qnn; sys_platform == "win32"

# Core numerical / data handling
# Pinned below 2.0: easyocr's torch build (2.2.2, the last version with
# Intel-Mac wheels) needs the numpy 1.x ABI.
numpy<2
pandas

# Screenshot OCR (feature 1: scam message detection from a screenshot)
# EasyOCR is pure-Python (no system Tesseract binary required) so it runs
# anywhere out of the box; swap for a Qualcomm AI Hub OCR model once available.
# opencv is pinned to match the numpy<2 constraint above.
pillow
easyocr
opencv-python-headless<4.10

# Text classification / lightweight local NLP (scam detection, spend categorization)
# skl2onnx exports the trained scikit-learn classifier to ONNX so it runs
# through the same QNN-aware runtime.py as the OCR models. langdetect flags
# non-English/gibberish input before it ever reaches the classifier.
scikit-learn
skl2onnx
langdetect

# Demo UI (app.py) — a minimal local web app, no login/accounts, so the
# on-device pipeline can be tried live without wiring up a real phone feed.
streamlit

# Dev tooling
pytest
