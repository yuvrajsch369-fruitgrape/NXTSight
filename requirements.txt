# NXTSight — on-device pipeline dependencies
# Target runtime: Snapdragon X Elite NPU (Hexagon) via Qualcomm AI Hub / ONNX Runtime QNN EP
#
# Every version below is pinned exactly (captured from a real working
# install, not guessed) so that `pip install -r requirements.txt` gives
# everyone the same environment instead of quietly resolving to whatever
# is newest on the day they happen to install it. If you need to bump a
# version, do it deliberately, re-run the full test suite, and re-pin —
# don't just delete the "==".

# Qualcomm AI Hub: compile + profile models for on-device NPU execution
# (used by scripts/aihub_*.py). qai-hub-models is intentionally NOT listed
# here — it's only needed for the one optional `python -m
# qai_hub_models.models.easyocr.export` CLI workflow (see README), and
# pulls in plain `opencv-python`, which silently conflicts with the
# `opencv-python-headless` that easyocr needs below (both install a `cv2`
# package at the same path; whichever installs second wins, non-
# deterministically). Install qai-hub-models separately only if you're
# running that specific export command.
qai-hub==0.55.0

# On-device inference runtime. onnxruntime-qnn ships ONNX Runtime with the
# QNN execution provider, which is how a compiled model actually reaches the
# Snapdragon Hexagon NPU — but it only publishes Windows/Linux-ARM wheels,
# and its version numbers do NOT track plain onnxruntime's (it's on its own
# 1.20.x/2.x line). Everywhere else (this dev Mac included) we install
# plain onnxruntime, and src/pipeline/runtime.py auto-detects which one is
# present at startup and falls back to the CPU provider when QNN isn't
# available. The onnxruntime-qnn pin below is UNVERIFIED on this machine
# (no Windows/ARM hardware to test on) — if it fails to install or import
# on the real Snapdragon device, check https://pypi.org/project/onnxruntime-qnn/
# for the current version list and adjust.
onnx==1.19.1  # newer onnx defaults to an IR version this onnxruntime can't load
onnxruntime==1.19.2; sys_platform != "win32"
onnxruntime-qnn==1.20.0; sys_platform == "win32"

# Core numerical / data handling.
# numpy pinned below 2.0: easyocr's torch build (2.2.2, the last version
# with Intel-Mac wheels — pinned explicitly below for the same reason, on
# every platform, so everyone gets the version this was actually tested
# against) needs the numpy 1.x ABI.
numpy==1.26.4
pandas==2.2.3

# Screenshot OCR (feature 1: scam message detection from a screenshot).
# EasyOCR is pure-Python (no system Tesseract binary required) so it runs
# anywhere out of the box. torch/opencv pinned explicitly (not just left to
# easyocr's own resolution) so every platform gets the exact combination
# this was tested against, not whatever's newest that day.
pillow==11.3.0
easyocr==1.7.2
torch==2.2.2
opencv-python-headless==4.9.0.80
joblib==1.6.0

# Text classification / lightweight local NLP (scam detection, spend
# categorization). skl2onnx exports the trained scikit-learn classifier to
# ONNX so it runs through the same QNN-aware runtime.py as the OCR models.
# langdetect flags non-English/gibberish input before it reaches a model.
scikit-learn==1.9.1
skl2onnx==1.20.0
langdetect==1.0.9

# Call Shield speech-to-text (feature 4: scam-call detection from an
# uploaded recording). Whisper's own audio loader shells out to a system
# `ffmpeg` binary for most formats, which isn't installed on every machine
# (this dev Mac included, no Homebrew either) — so src/pipeline/stt.py
# decodes WAV files itself (stdlib `wave` + numpy, already a dependency
# above) and only uses Whisper for the actual transcription, sidestepping
# the ffmpeg dependency entirely. Already reuses torch, pinned above.
#
# numba is pinned explicitly even though we never import it ourselves:
# openai-whisper depends on it with NO version constraint at all, so pip
# picks whatever's newest — which pulls in a newer llvmlite than 0.43.0,
# the last version with an Intel-Mac wheel (same story as the torch pin
# above). Without this pin, `pip install -r requirements.txt` tries to
# compile llvmlite from source on Intel Macs and fails outright unless a
# full LLVM toolchain happens to be installed. Confirmed reproducible on a
# genuinely clean venv, not a one-off — this isn't hypothetical.
numba==0.60.0
openai-whisper==20250625

# Demo UI (app.py) — a minimal local web app, no login/accounts, so the
# on-device pipeline can be tried live without wiring up a real phone feed.
streamlit==1.63.0

# TLS root certificates. Needed by EasyOCR's one-time model download and
# by pip itself on some setups — pinned here so the macOS
# CERTIFICATE_VERIFY_FAILED workaround in the README always has it
# available, no separate manual install step required.
certifi==2026.7.22

# Dev tooling
pytest==9.1.1
